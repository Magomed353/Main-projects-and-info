    # @authorization
    @staticmethod 
    async def authorization(login, password):
        password_hash = await UserService.get_hash_us(login)
        if password_hash == None:
            return JSONResponse(
                status_code=status.HTTP_401_UNAUTHORIZED,
                content={"status": False, "message": "Неверный логин"}
                )
        result = bcrypt.checkpw(password.encode("utf-8"), password_hash.encode("utf-8")) 
        if result == False:
            return JSONResponse(
                status_code=status.HTTP_401_UNAUTHORIZED,
                content={"status": False, "message": "Неверный пароль"}
                )
        return {"status": True, "message": "Авторизованы"}

    # @registration
    @staticmethod
    async def registration(login, password):
        if await UserService.get_login_us(login) != None:
            return JSONResponse(
                status_code=status.HTTP_400_BAD_REQUEST,
                content={"status": False, "message":"Логин уже существует"}
            )
        password_hash = bcrypt.hashpw(password.encode("utf-8"),bcrypt.gensalt()).decode("utf-8")
        new_user = UserSchema(login=login, password_hash=password_hash)
        await UserService.add_user(new_user)
